/* =========================================================
   11A. STANDARD regid / uid COLUMNS
   All Critical Workflow tables keep:
       regid INT NULL
       uid   INT NULL

   This block also manages databases where an older version
   of these tables was already created.
   ========================================================= */

IF OBJECT_ID('dbo.Pms_tbl_CriticalLabel','U') IS NOT NULL
BEGIN
    IF COL_LENGTH('dbo.Pms_tbl_CriticalLabel','regid') IS NULL
        ALTER TABLE dbo.Pms_tbl_CriticalLabel ADD regid INT NULL;
    ELSE
    BEGIN
        DECLARE @dfCriticalLabelRegid NVARCHAR(128);
        SELECT @dfCriticalLabelRegid = dc.name
        FROM sys.default_constraints dc
        INNER JOIN sys.columns c
            ON c.default_object_id = dc.object_id
        WHERE dc.parent_object_id = OBJECT_ID('dbo.Pms_tbl_CriticalLabel')
          AND c.name = 'regid';

        IF @dfCriticalLabelRegid IS NOT NULL
            EXEC('ALTER TABLE dbo.Pms_tbl_CriticalLabel DROP CONSTRAINT [' + @dfCriticalLabelRegid + ']');

        ALTER TABLE dbo.Pms_tbl_CriticalLabel ALTER COLUMN regid INT NULL;
    END

    IF COL_LENGTH('dbo.Pms_tbl_CriticalLabel','uid') IS NULL
        ALTER TABLE dbo.Pms_tbl_CriticalLabel ADD uid INT NULL;
END
GO

/* 13. Master data */
IF OBJECT_ID('dbo.usp_Pms_Critical_GetMasters','P') IS NOT NULL
    DROP PROCEDURE dbo.usp_Pms_Critical_GetMasters;
GO

CREATE PROCEDURE dbo.usp_Pms_Critical_GetMasters
(
    @regid INT,
    @uid INT = NULL
)
AS
BEGIN
    SET NOCOUNT ON;

    /* ---------------------------------------------------------
       COMPANY CRITICAL LABELS
       Default rows have regid NULL and work only as templates.
       Copy each missing template into the logged-in company so
       Edit/Update always works against regid=@regid.
       --------------------------------------------------------- */

    INSERT INTO dbo.Pms_tbl_CriticalLabel
    (
        CriticalLabel,
        ColorCode,
        Severity,
        DisplayOrder,
        regid,
        uid,
        status,
        archive
    )
    SELECT
        G.CriticalLabel,
        G.ColorCode,
        G.Severity,
        G.DisplayOrder,
        @regid,
        @uid,
        1,
        0
    FROM dbo.Pms_tbl_CriticalLabel G
    WHERE G.regid IS NULL
      AND G.status=1
      AND G.archive=0
      AND NOT EXISTS
      (
          SELECT 1
          FROM dbo.Pms_tbl_CriticalLabel C
          WHERE C.regid=@regid
            AND C.status=1
            AND C.archive=0
            AND C.CriticalLabel=G.CriticalLabel
      );

    /* Company Critical Labels only */
    SELECT
        id,
        CriticalLabel,
        ColorCode,
        Severity,
        DisplayOrder
    FROM dbo.Pms_tbl_CriticalLabel
    WHERE status=1
      AND archive=0
      AND regid=@regid
    ORDER BY DisplayOrder,Severity,CriticalLabel;

    /* Apply Levels */
    SELECT
        id,
        LevelName,
        LevelCode
    FROM dbo.Pms_tbl_CriticalLevelMaster
    WHERE status=1
      AND archive=0
    ORDER BY DisplayOrder;

    /* Condition Fields */
    SELECT
        id,
        FieldName,
        FieldKey,
        DataType,
        SourceType
    FROM dbo.Pms_tbl_CriticalFieldMaster
    WHERE status=1
      AND archive=0
    ORDER BY DisplayOrder;

    /* Operators */
    SELECT
        id,
        OperatorName,
        OperatorSymbol
    FROM dbo.Pms_tbl_CriticalOperatorMaster
    WHERE status=1
      AND archive=0
    ORDER BY DisplayOrder;

    /* Logical Operators */
    SELECT
        id,
        OperatorName
    FROM dbo.Pms_tbl_CriticalLogicalOperator
    WHERE status=1
      AND archive=0
    ORDER BY DisplayOrder;

    /* Actions */
    SELECT
        id,
        ActionName,
        ActionCode
    FROM dbo.Pms_tbl_CriticalActionMaster
    WHERE status=1
      AND archive=0
    ORDER BY DisplayOrder;

    /* Existing PMS Status Master - current company only */
    SELECT
        id AS ValueId,
        ProjectStatus AS ValueText
    FROM dbo.Pms_tbl_Projectstatus
    WHERE status=1
      AND archive=0
      AND regid=@regid
    ORDER BY ProjectStatus;

    /* Existing PMS Priority Master - current company only */
    SELECT
        id AS ValueId,
        name AS ValueText
    FROM dbo.Pms_tbl_Priority
    WHERE status=1
      AND archive=0
      AND regid=@regid
    ORDER BY name;
END
GO

GO

CREATE PROCEDURE dbo.usp_Pms_Critical_GetFieldValues
(
    @FieldId INT,
    @regid INT
)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @SourceType NVARCHAR(50);

    SELECT
        @SourceType=SourceType
    FROM dbo.Pms_tbl_CriticalFieldMaster
    WHERE id=@FieldId
      AND status=1
      AND archive=0;

    IF @SourceType='PRIORITY'
    BEGIN
        SELECT
            id AS ValueId,
            name AS ValueText
        FROM dbo.Pms_tbl_Priority
        WHERE status=1
          AND archive=0
          AND regid=@regid
        ORDER BY name;

        RETURN;
    END

    IF @SourceType='STATUS'
    BEGIN
        SELECT
            id AS ValueId,
            ProjectStatus AS ValueText
        FROM dbo.Pms_tbl_Projectstatus
        WHERE status=1
          AND archive=0
          AND regid=@regid
        ORDER BY ProjectStatus;

        RETURN;
    END

    SELECT
        CAST(NULL AS INT) AS ValueId,
        CAST(NULL AS NVARCHAR(200)) AS ValueText
    WHERE 1=0;
END
GO

/* 15. Save label */
IF OBJECT_ID('dbo.usp_Pms_CriticalLabel_Save','P') IS NOT NULL
    DROP PROCEDURE dbo.usp_Pms_CriticalLabel_Save;
GO

CREATE PROCEDURE dbo.usp_Pms_CriticalLabel_Save
(
    @id INT=0,
    @CriticalLabel NVARCHAR(100),
    @ColorCode NVARCHAR(20),
    @Severity INT,
    @DisplayOrder INT=0,
    @regid INT,
    @uid INT=NULL
)
AS
BEGIN
    SET NOCOUNT ON;

    IF ISNULL(LTRIM(RTRIM(@CriticalLabel)),'')=''
    BEGIN
        RAISERROR('Critical label is required.',16,1);
        RETURN;
    END

    IF EXISTS
    (
        SELECT 1
        FROM dbo.Pms_tbl_CriticalLabel
        WHERE CriticalLabel=@CriticalLabel
          AND regid=@regid
          AND archive=0
          AND id<>ISNULL(@id,0)
    )
    BEGIN
        RAISERROR('Critical label already exists.',16,1);
        RETURN;
    END

    IF ISNULL(@id,0)=0
    BEGIN
        INSERT INTO dbo.Pms_tbl_CriticalLabel
        (
            CriticalLabel,
            ColorCode,
            Severity,
            DisplayOrder,
            regid,
            uid,
            status,
            archive
        )
        VALUES
        (
            @CriticalLabel,
            ISNULL(NULLIF(@ColorCode,''),'#28a745'),
            ISNULL(@Severity,1),
            ISNULL(@DisplayOrder,0),
            @regid,
            @uid,
            1,
            0
        );

        SELECT
            CAST(SCOPE_IDENTITY() AS INT) AS id,
            'Saved successfully.' AS Message;

        RETURN;
    END

    UPDATE dbo.Pms_tbl_CriticalLabel
    SET
        CriticalLabel=@CriticalLabel,
        ColorCode=ISNULL(NULLIF(@ColorCode,''),'#28a745'),
        Severity=ISNULL(@Severity,1),
        DisplayOrder=ISNULL(@DisplayOrder,0),
        uid=@uid,
        modifieddate=GETDATE()
    WHERE id=@id
      AND regid=@regid
      AND archive=0;

    IF @@ROWCOUNT=0
    BEGIN

/* 16. Save / update rule via XML */
IF OBJECT_ID('dbo.usp_Pms_CriticalRule_Save','P') IS NOT NULL DROP PROCEDURE dbo.usp_Pms_CriticalRule_Save;
GO
CREATE PROCEDURE dbo.usp_Pms_CriticalRule_Save
    @RuleId INT=0,
    @RuleName NVARCHAR(250),
    @LevelId INT,
    @CriticalLabelId INT,
    @PriorityId INT=NULL,
    @RulePriority INT=100,
    @StopProcessing BIT=1,
    @Description NVARCHAR(1000)=NULL,
    @regid INT,
    @uid INT=NULL,
    @ConditionsXml XML,
    @ActionsXml XML=NULL
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        IF ISNULL(LTRIM(RTRIM(@RuleName)),'')='' RAISERROR('Rule name is required.',16,1);
        IF NOT EXISTS(SELECT 1 FROM dbo.Pms_tbl_CriticalLevelMaster WHERE id=@LevelId AND status=1 AND archive=0)
            RAISERROR('Invalid apply level.',16,1);
        IF NOT EXISTS(SELECT 1 FROM dbo.Pms_tbl_CriticalLabel WHERE id=@CriticalLabelId AND status=1 AND archive=0)
            RAISERROR('Invalid critical label.',16,1);
        IF @ConditionsXml IS NULL OR @ConditionsXml.exist('/Conditions/Condition')=0
            RAISERROR('At least one condition is required.',16,1);

        IF ISNULL(@RuleId,0)=0
        BEGIN
            INSERT dbo.Pms_tbl_CriticalRule
            (
                RuleName,LevelId,CriticalLabelId,PriorityId,RulePriority,StopProcessing,
                Description,regid,uid,status,archive
            )
            VALUES
            (
                @RuleName,@LevelId,@CriticalLabelId,@PriorityId,ISNULL(@RulePriority,100),
                ISNULL(@StopProcessing,1),@Description,@regid,@uid,1,0
            );
            SET @RuleId=CAST(SCOPE_IDENTITY() AS INT);
        END
        ELSE
        BEGIN
            UPDATE dbo.Pms_tbl_CriticalRule
            SET RuleName=@RuleName,
                LevelId=@LevelId,
                CriticalLabelId=@CriticalLabelId,
                PriorityId=@PriorityId,
                RulePriority=ISNULL(@RulePriority,100),
                StopProcessing=ISNULL(@StopProcessing,1),
                Description=@Description,
                uid=@uid,
                modifieddate=GETDATE()
            WHERE id=@RuleId AND regid=@regid AND archive=0;

            IF @@ROWCOUNT=0 RAISERROR('Critical rule not found.',16,1);

            DELETE dbo.Pms_tbl_CriticalRuleCondition WHERE RuleId=@RuleId;
            DELETE dbo.Pms_tbl_CriticalRuleAction WHERE RuleId=@RuleId;
        END

        INSERT dbo.Pms_tbl_CriticalRuleCondition
        (
            RuleId,SequenceNo,OpenBracket,FieldId,OperatorId,CompareValue,
            CompareValue2,LogicalOperatorId,CloseBracket,GroupNo,regid,uid,status,archive
        )
        SELECT
            @RuleId,
            T.C.value('@SequenceNo','INT'),
            NULLIF(T.C.value('@OpenBracket','NVARCHAR(10)'),''),
            T.C.value('@FieldId','INT'),
            T.C.value('@OperatorId','INT'),
            NULLIF(T.C.value('@CompareValue','NVARCHAR(500)'),''),
            NULLIF(T.C.value('@CompareValue2','NVARCHAR(500)'),''),
            TRY_CONVERT(INT,NULLIF(T.C.value('@LogicalOperatorId','NVARCHAR(20)'),''))
            ,NULLIF(T.C.value('@CloseBracket','NVARCHAR(10)'),''),
            ISNULL(TRY_CONVERT(INT,NULLIF(T.C.value('@GroupNo','NVARCHAR(20)'),'')),1),
            @regid,@uid,
            1,0
        FROM @ConditionsXml.nodes('/Conditions/Condition') T(C);

        IF @ActionsXml IS NOT NULL AND @ActionsXml.exist('/Actions/Action')=1
        BEGIN
            INSERT dbo.Pms_tbl_CriticalRuleAction
            (
                RuleId,ActionId,ActionValue,RecipientType,SequenceNo,regid,uid,status,archive
            )
            SELECT
                @RuleId,
                T.C.value('@ActionId','INT'),
                NULLIF(T.C.value('@ActionValue','NVARCHAR(1000)'),''),
                NULLIF(T.C.value('@RecipientType','NVARCHAR(100)'),''),
                ISNULL(TRY_CONVERT(INT,NULLIF(T.C.value('@SequenceNo','NVARCHAR(20)'),'')),1),
                @regid,@uid,
                1,0
            FROM @ActionsXml.nodes('/Actions/Action') T(C);
        END

        COMMIT;
        SELECT @RuleId RuleId,'Rule saved successfully.' Message;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT>0 ROLLBACK;
        DECLARE @Err NVARCHAR(4000)=ERROR_MESSAGE();
        RAISERROR(@Err,16,1);
    END CATCH
END
GO

CREATE PROCEDURE dbo.usp_Pms_CriticalResult_Save
    @EntityType NVARCHAR(30),
    @EntityId INT,
    @ProjectId INT=NULL,
    @MilestoneId INT=NULL,
    @TaskId INT=NULL,
    @SubTaskId INT=NULL,
    @CriticalLabelId INT,
    @MatchedRuleId INT=NULL,
    @PriorityId INT=NULL,
    @ProjectStatusId INT=NULL,
    @TimelinePercent DECIMAL(10,2)=NULL,
    @CompletionPercent DECIMAL(10,2)=NULL,
    @DurationPercent DECIMAL(10,2)=NULL,
    @OverdueDays INT=NULL,
    @PlannedDays DECIMAL(10,2)=NULL,
    @AllocatedHours DECIMAL(18,2)=NULL,
    @SpentHours DECIMAL(18,2)=NULL,
    @CriticalChildCount INT=NULL,
    @CriticalChildPercent DECIMAL(10,2)=NULL,
    @Reason NVARCHAR(MAX)=NULL,
    @regid INT,
    @uid INT=NULL
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

        FROM dbo.Pms_tbl_CriticalResult
        WHERE regid=@regid AND EntityType=@EntityType AND EntityId=@EntityId
          AND status=1 AND archive=0
        ORDER BY id DESC;

        UPDATE dbo.Pms_tbl_CriticalResult
        SET status=0,archive=1
        WHERE regid=@regid AND EntityType=@EntityType AND EntityId=@EntityId
          AND status=1 AND archive=0;

        INSERT dbo.Pms_tbl_CriticalResult
        (
            EntityType,EntityId,ProjectId,MilestoneId,TaskId,SubTaskId,
            CriticalLabelId,MatchedRuleId,PriorityId,ProjectStatusId,
            TimelinePercent,CompletionPercent,DurationPercent,OverdueDays,
            PlannedDays,AllocatedHours,SpentHours,CriticalChildCount,CriticalChildPercent,

        )

            @CriticalLabelId,@MatchedRuleId,@PriorityId,@ProjectStatusId,

            @PlannedDays,@AllocatedHours,@SpentHours,@CriticalChildCount,@CriticalChildPercent,
            @Reason,@regid,@uid,1,0
        );

        IF ISNULL(@OldCriticalLabelId,0)<>ISNULL(@CriticalLabelId,0)
        BEGIN
            INSERT dbo.Pms_tbl_CriticalHistory
            (
                EntityType,EntityId,ProjectId,MilestoneId,TaskId,SubTaskId,
                RuleId,OldCriticalLabelId,NewCriticalLabelId,
                TimelinePercent,CompletionPercent,DurationPercent,OverdueDays,
                Reason,regid,uid

                @EntityType,@EntityId,@ProjectId,@MilestoneId,@TaskId,@SubTaskId,
                @MatchedRuleId,@OldCriticalLabelId,@CriticalLabelId,
                @TimelinePercent,@CompletionPercent,@DurationPercent,@OverdueDays,
                @Reason,@regid,@uid
            );
        END

        COMMIT;
        SELECT 1 Result,'Critical result updated successfully.' Message;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT>0 ROLLBACK;
        DECLARE @Err NVARCHAR(4000)=ERROR_MESSAGE();
        RAISERROR(@Err,16,1);
    END CATCH
END

    WHERE O.OperatorSymbol=R.OperatorSymbol AND O.archive=0
);
GO

   4. Master loader - separate Project/Milestone/Task/SubTask.
   Result-set order remains compatible with current C# page.
   --------------------------------------------------------- */
IF OBJECT_ID('dbo.usp_Pms_Critical_GetMasters','P') IS NOT NULL
    DROP PROCEDURE dbo.usp_Pms_Critical_GetMasters;
GO
CREATE PROCEDURE dbo.usp_Pms_Critical_GetMasters
(
    @regid INT,
    @uid INT=NULL
)
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS(SELECT 1 FROM dbo.Pms_tbl_CriticalLabel WHERE regid=@regid)
    BEGIN
        INSERT dbo.Pms_tbl_CriticalLabel
        (CriticalLabel,ColorCode,Severity,DisplayOrder,regid,uid,status,archive)
        SELECT G.CriticalLabel,G.ColorCode,G.Severity,G.DisplayOrder,@regid,@uid,1,0
        FROM dbo.Pms_tbl_CriticalLabel G
        WHERE G.regid IS NULL AND G.status=1 AND G.archive=0;
    END

    /* 1 labels */
    SELECT id,CriticalLabel,ColorCode,Severity,DisplayOrder
    FROM dbo.Pms_tbl_CriticalLabel
    WHERE status=1 AND archive=0 AND regid=@regid
    ORDER BY DisplayOrder,Severity,CriticalLabel;

    /* 2 levels */
    SELECT id,LevelName,LevelCode
    FROM dbo.Pms_tbl_CriticalLevelMaster
    WHERE status=1 AND archive=0
      AND LevelCode IN ('PROJECT','MILESTONE','TASK','SUBTASK')
    ORDER BY CASE LevelCode
        WHEN 'PROJECT' THEN 1
        WHEN 'MILESTONE' THEN 2
        WHEN 'TASK' THEN 3
        WHEN 'SUBTASK' THEN 4
        ELSE 99 END;

    /* 3 self fields */
    SELECT id,FieldName,FieldKey,DataType,SourceType
    FROM dbo.Pms_tbl_CriticalFieldMaster
    WHERE status=1 AND archive=0
      AND FieldKey IN
      ('STATUS','PRIORITY','COMPLETION_PERCENT','DURATION_PERCENT','PROGRESS_DIFFERENCE_PERCENT')
    ORDER BY DisplayOrder;

    /* 4 operators */
    SELECT id,OperatorName,OperatorSymbol
    FROM dbo.Pms_tbl_CriticalOperatorMaster
    WHERE status=1 AND archive=0
      AND OperatorSymbol IN ('=','!=','>','<','>=','<=','BETWEEN','IN','NOT IN')
    ORDER BY DisplayOrder;

    /* 5 logical operators */
    SELECT id,OperatorName
    FROM dbo.Pms_tbl_CriticalLogicalOperator
    WHERE status=1 AND archive=0
    ORDER BY DisplayOrder;

    /* 6 legacy actions - retained only for current C# result-set indexes */
    SELECT id,ActionName,ActionCode
    FROM dbo.Pms_tbl_CriticalActionMaster
    WHERE status=1 AND archive=0
    ORDER BY DisplayOrder;

    /* 7 status */
    SELECT id AS ValueId,ProjectStatus AS ValueText
    FROM dbo.Pms_tbl_Projectstatus
    WHERE status=1 AND archive=0 AND regid=@regid
    ORDER BY ProjectStatus;

    /* 8 priority */
    SELECT id AS ValueId,name AS ValueText
    FROM dbo.Pms_tbl_Priority
    WHERE status=1 AND archive=0 AND regid=@regid
    ORDER BY name;
END
GO

/* ---------------------------------------------------------
   5. Save/update hierarchy-aware rule.
   --------------------------------------------------------- */
IF OBJECT_ID('dbo.usp_Pms_CriticalRule_Save','P') IS NOT NULL
    DROP PROCEDURE dbo.usp_Pms_CriticalRule_Save;
GO
CREATE PROCEDURE dbo.usp_Pms_CriticalRule_Save
    @RuleId INT=0,
    @RuleName NVARCHAR(250),
    @LevelId INT,
    @CriticalLabelId INT,
    @PriorityId INT=NULL,              -- legacy; UI sends NULL
    @RulePriority INT=1,
    @StopProcessing BIT=1,
    @Description NVARCHAR(1000)=NULL,
    @regid INT,
    @uid INT=NULL,
    @ConditionsXml XML,
    @ActionsXml XML=NULL               -- legacy compatibility
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        DECLARE @ParentLevelCode NVARCHAR(30);
        SELECT @ParentLevelCode=LevelCode
        FROM dbo.Pms_tbl_CriticalLevelMaster
        WHERE id=@LevelId AND status=1 AND archive=0;

        IF ISNULL(LTRIM(RTRIM(@RuleName)),'')=''
            RAISERROR('Rule Name is required.',16,1);

        IF ISNULL(@RulePriority,0)<1
            RAISERROR('Rule Order must be 1 or greater.',16,1);

        IF @ParentLevelCode NOT IN ('PROJECT','MILESTONE','TASK','SUBTASK')
            RAISERROR('Apply On must be Project, Milestone, Task or Sub Task.',16,1);

        IF NOT EXISTS
        (
            SELECT 1 FROM dbo.Pms_tbl_CriticalLabel
            WHERE id=@CriticalLabelId AND regid=@regid AND status=1 AND archive=0
        )
            RAISERROR('Invalid Critical Label.',16,1);

        IF @ConditionsXml IS NULL OR @ConditionsXml.exist('/Conditions/Condition')=0
            RAISERROR('At least one condition is required.',16,1);

        /* ConditionSource must be SELF or CHILD. Empty old input is treated as SELF. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode
            ) S
            WHERE S.SourceCode NOT IN ('SELF','CHILD')
        )
            RAISERROR('Invalid condition source.',16,1);

        /* Every condition must use an active operator. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            WHERE NOT EXISTS
            (
                SELECT 1
                FROM dbo.Pms_tbl_CriticalOperatorMaster O
                WHERE O.id=TRY_CONVERT(INT,NULLIF(T.C.value('@OperatorId','NVARCHAR(30)'),''))
                  AND O.status=1 AND O.archive=0
            )
        )
            RAISERROR('One or more condition operators are invalid.',16,1);

        /* SELF field validation. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@FieldId','NVARCHAR(30)'),'')) AS FieldId
            ) X
            WHERE X.SourceCode='SELF'
              AND NOT EXISTS
              (
                  SELECT 1 FROM dbo.Pms_tbl_CriticalFieldMaster F
                  WHERE F.id=X.FieldId AND F.status=1 AND F.archive=0
                    AND F.FieldKey IN
                    ('STATUS','PRIORITY','COMPLETION_PERCENT','DURATION_PERCENT','PROGRESS_DIFFERENCE_PERCENT')
              )
        )
            RAISERROR('One or more Self condition fields are invalid.',16,1);

        /* SELF operator validation. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@FieldId','NVARCHAR(30)'),'')) AS FieldId,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@OperatorId','NVARCHAR(30)'),'')) AS OperatorId
            ) X
            INNER JOIN dbo.Pms_tbl_CriticalFieldMaster F ON F.id=X.FieldId
            INNER JOIN dbo.Pms_tbl_CriticalOperatorMaster O ON O.id=X.OperatorId
            WHERE X.SourceCode='SELF'
              AND
              (
                  (F.DataType='MASTER' AND O.OperatorSymbol NOT IN ('=','!=','IN','NOT IN'))
                  OR
                  (F.DataType='NUMBER' AND O.OperatorSymbol NOT IN ('=','!=','>','<','>=','<=','BETWEEN'))
              )
        )
            RAISERROR('Selected operator is not valid for the selected Self field.',16,1);

        /* CHILD is not valid on SubTask. */
        IF @ParentLevelCode='SUBTASK'
           AND EXISTS
           (
               SELECT 1 FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
               WHERE UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF'))='CHILD'
           )
            RAISERROR('Sub Task cannot contain Child conditions.',16,1);

        /* CHILD level/label/measure/operator validation. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@ChildLevelId','NVARCHAR(30)'),'')) AS ChildLevelId,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@ChildCriticalLabelId','NVARCHAR(30)'),'')) AS ChildLabelId,
                       UPPER(NULLIF(T.C.value('@ChildMeasure','NVARCHAR(30)'),'')) AS ChildMeasure,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@OperatorId','NVARCHAR(30)'),'')) AS OperatorId
            ) X
            LEFT JOIN dbo.Pms_tbl_CriticalLevelMaster CLV ON CLV.id=X.ChildLevelId AND CLV.status=1 AND CLV.archive=0
            LEFT JOIN dbo.Pms_tbl_CriticalLabel CL ON CL.id=X.ChildLabelId AND CL.regid=@regid AND CL.status=1 AND CL.archive=0
            LEFT JOIN dbo.Pms_tbl_CriticalOperatorMaster O ON O.id=X.OperatorId AND O.status=1 AND O.archive=0
            WHERE X.SourceCode='CHILD'
              AND
              (
                  CLV.id IS NULL
                  OR CL.id IS NULL
                  OR X.ChildMeasure NOT IN ('COUNT','PERCENTAGE')
                  OR O.OperatorSymbol NOT IN ('=','!=','>','<','>=','<=','BETWEEN')
                  OR (@ParentLevelCode='PROJECT' AND CLV.LevelCode NOT IN ('MILESTONE','TASK','SUBTASK'))
                  OR (@ParentLevelCode='MILESTONE' AND CLV.LevelCode NOT IN ('TASK','SUBTASK'))
                  OR (@ParentLevelCode='TASK' AND CLV.LevelCode<>'SUBTASK')
              )
        )
            RAISERROR('One or more Child conditions are invalid for the selected Apply On level.',16,1);

        /* Value required for every condition. */
        IF EXISTS
        (
            SELECT 1 FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            WHERE ISNULL(LTRIM(RTRIM(T.C.value('@CompareValue','NVARCHAR(500)'))),'')=''
        )
            RAISERROR('Condition value is required.',16,1);

        /* BETWEEN second value required. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            INNER JOIN dbo.Pms_tbl_CriticalOperatorMaster O
                ON O.id=TRY_CONVERT(INT,NULLIF(T.C.value('@OperatorId','NVARCHAR(30)'),''))
            WHERE O.OperatorSymbol='BETWEEN'
              AND ISNULL(LTRIM(RTRIM(T.C.value('@CompareValue2','NVARCHAR(500)'))),'')=''
        )
            RAISERROR('Second value is required for BETWEEN.',16,1);

        /* Numeric SELF values. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@FieldId','NVARCHAR(30)'),'')) AS FieldId
            ) X
            INNER JOIN dbo.Pms_tbl_CriticalFieldMaster F ON F.id=X.FieldId
            WHERE X.SourceCode='SELF' AND F.DataType='NUMBER'
              AND TRY_CONVERT(DECIMAL(18,4),T.C.value('@CompareValue','NVARCHAR(500)')) IS NULL
        )
            RAISERROR('Numeric Self condition contains an invalid value.',16,1);

        /* Numeric SELF BETWEEN second value must also be numeric. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@FieldId','NVARCHAR(30)'),'')) AS FieldId,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@OperatorId','NVARCHAR(30)'),'')) AS OperatorId
            ) X
            INNER JOIN dbo.Pms_tbl_CriticalFieldMaster F ON F.id=X.FieldId
            INNER JOIN dbo.Pms_tbl_CriticalOperatorMaster O ON O.id=X.OperatorId
            WHERE X.SourceCode='SELF' AND F.DataType='NUMBER' AND O.OperatorSymbol='BETWEEN'
              AND TRY_CONVERT(DECIMAL(18,4),T.C.value('@CompareValue2','NVARCHAR(500)')) IS NULL
        )
            RAISERROR('Numeric Self BETWEEN second value is invalid.',16,1);

        /* CHILD values are numeric; percentage must be 0..100. */
        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode,
                       UPPER(NULLIF(T.C.value('@ChildMeasure','NVARCHAR(30)'),'')) AS ChildMeasure,
                       TRY_CONVERT(DECIMAL(18,4),T.C.value('@CompareValue','NVARCHAR(500)')) AS NumValue
            ) X
            WHERE X.SourceCode='CHILD'
              AND (X.NumValue IS NULL OR X.NumValue<0 OR (X.ChildMeasure='PERCENTAGE' AND X.NumValue>100))
        )
            RAISERROR('Child Count/Percentage value is invalid.',16,1);

        IF EXISTS
        (
            SELECT 1
            FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            CROSS APPLY
            (
                SELECT UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')) AS SourceCode,
                       UPPER(NULLIF(T.C.value('@ChildMeasure','NVARCHAR(30)'),'')) AS ChildMeasure,
                       TRY_CONVERT(DECIMAL(18,4),T.C.value('@CompareValue2','NVARCHAR(500)')) AS NumValue2,
                       TRY_CONVERT(INT,NULLIF(T.C.value('@OperatorId','NVARCHAR(30)'),'')) AS OperatorId
            ) X
            INNER JOIN dbo.Pms_tbl_CriticalOperatorMaster O ON O.id=X.OperatorId
            WHERE X.SourceCode='CHILD' AND O.OperatorSymbol='BETWEEN'
              AND (X.NumValue2 IS NULL OR X.NumValue2<0 OR (X.ChildMeasure='PERCENTAGE' AND X.NumValue2>100))
        )
            RAISERROR('Child BETWEEN second value is invalid.',16,1);

        IF ISNULL(@RuleId,0)=0
        BEGIN
            INSERT dbo.Pms_tbl_CriticalRule
            (RuleName,LevelId,CriticalLabelId,PriorityId,RulePriority,StopProcessing,Description,regid,uid,status,archive)
            VALUES(@RuleName,@LevelId,@CriticalLabelId,NULL,@RulePriority,ISNULL(@StopProcessing,1),@Description,@regid,@uid,1,0);
            SET @RuleId=CAST(SCOPE_IDENTITY() AS INT);
        END
        ELSE
        BEGIN
            UPDATE dbo.Pms_tbl_CriticalRule
            SET RuleName=@RuleName,
                LevelId=@LevelId,
                CriticalLabelId=@CriticalLabelId,
                PriorityId=NULL,
                RulePriority=@RulePriority,
                StopProcessing=ISNULL(@StopProcessing,1),
                Description=@Description,
                uid=@uid,
                modifieddate=GETDATE()
            WHERE id=@RuleId AND regid=@regid AND archive=0;

            IF @@ROWCOUNT=0 RAISERROR('Critical rule not found.',16,1);

            DELETE dbo.Pms_tbl_CriticalRuleCondition WHERE RuleId=@RuleId;
            DELETE dbo.Pms_tbl_CriticalRuleAction WHERE RuleId=@RuleId;
        END

        INSERT dbo.Pms_tbl_CriticalRuleCondition
        (
            RuleId,SequenceNo,ConditionSource,OpenBracket,FieldId,
            ChildLevelId,ChildCriticalLabelId,ChildMeasure,
            OperatorId,CompareValue,CompareValue2,LogicalOperatorId,
            CloseBracket,GroupNo,regid,uid,status,archive
        )
        SELECT
            @RuleId,
            T.C.value('@SequenceNo','INT'),
            UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')),
            NULLIF(T.C.value('@OpenBracket','NVARCHAR(10)'),''),
            CASE WHEN UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF'))='SELF'
                 THEN TRY_CONVERT(INT,NULLIF(T.C.value('@FieldId','NVARCHAR(30)'),'')) ELSE NULL END,
            CASE WHEN UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF'))='CHILD'
                 THEN TRY_CONVERT(INT,NULLIF(T.C.value('@ChildLevelId','NVARCHAR(30)'),'')) ELSE NULL END,
            CASE WHEN UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF'))='CHILD'
                 THEN TRY_CONVERT(INT,NULLIF(T.C.value('@ChildCriticalLabelId','NVARCHAR(30)'),'')) ELSE NULL END,
            CASE WHEN UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF'))='CHILD'
                 THEN UPPER(NULLIF(T.C.value('@ChildMeasure','NVARCHAR(30)'),'')) ELSE NULL END,
            T.C.value('@OperatorId','INT'),
            NULLIF(T.C.value('@CompareValue','NVARCHAR(500)'),''),
            NULLIF(T.C.value('@CompareValue2','NVARCHAR(500)'),''),
            TRY_CONVERT(INT,NULLIF(T.C.value('@LogicalOperatorId','NVARCHAR(20)'),'')),
            NULLIF(T.C.value('@CloseBracket','NVARCHAR(10)'),''),
            ISNULL(TRY_CONVERT(INT,NULLIF(T.C.value('@GroupNo','NVARCHAR(20)'),'')),1),
            @regid,@uid,1,0
        FROM @ConditionsXml.nodes('/Conditions/Condition') T(C);

        COMMIT;
        SELECT @RuleId AS RuleId,'Rule saved successfully.' AS Message;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT>0 ROLLBACK;
        DECLARE @Err NVARCHAR(4000)=ERROR_MESSAGE();
        RAISERROR(@Err,16,1);
    END CATCH
END
GO

   --------------------------------------------------------- */

    DROP PROCEDURE dbo.usp_Pms_CriticalLabel_Delete;
GO
CREATE PROCEDURE dbo.usp_Pms_CriticalLabel_Delete
(
    @id INT,
    @regid INT,
    @uid INT=NULL
)
AS
BEGIN
    SET NOCOUNT ON;

    IF ISNULL(@id,0)=0
    BEGIN
        SELECT 0 AS Result,'Invalid critical label.' AS Message;
        RETURN;
    END

    IF NOT EXISTS
    (
        SELECT 1 FROM dbo.Pms_tbl_CriticalLabel
        WHERE id=@id AND regid=@regid AND archive=0
    )
    BEGIN
        SELECT 0 AS Result,'Critical label not found for current company.' AS Message;
        RETURN;
    END

    IF EXISTS
    (
        SELECT 1 FROM dbo.Pms_tbl_CriticalRule
        WHERE CriticalLabelId=@id AND regid=@regid AND archive=0
    )
    BEGIN
        SELECT 0 AS Result,'This Critical Label is used as a rule result. Change or delete that rule first.' AS Message;
        RETURN;
    END

    IF EXISTS
    (
        SELECT 1
        FROM dbo.Pms_tbl_CriticalRuleCondition C
        INNER JOIN dbo.Pms_tbl_CriticalRule R ON R.id=C.RuleId
        WHERE C.ChildCriticalLabelId=@id
          AND R.regid=@regid
          AND R.archive=0
          AND C.archive=0
    )
    BEGIN
        SELECT 0 AS Result,'This Critical Label is used in a Child condition. Change or delete that rule first.' AS Message;
        RETURN;
    END

    UPDATE dbo.Pms_tbl_CriticalLabel
    SET status=0,archive=1,uid=@uid,modifieddate=GETDATE()
    WHERE id=@id AND regid=@regid AND archive=0;

    IF @@ROWCOUNT>0 SELECT 1 AS Result,'Critical label deleted successfully.' AS Message;
    ELSE SELECT 0 AS Result,'Critical label could not be deleted.' AS Message;
END
GO

/* ---------------------------------------------------------
   9. Active rules for runtime evaluator.
   Returns Self and Child condition metadata.
   --------------------------------------------------------- */
IF OBJECT_ID('dbo.usp_Pms_Critical_GetActiveRules','P') IS NOT NULL
    DROP PROCEDURE dbo.usp_Pms_Critical_GetActiveRules;
GO
CREATE PROCEDURE dbo.usp_Pms_Critical_GetActiveRules
    @regid INT,
    @LevelCode NVARCHAR(30)
AS
BEGIN
    SET NOCOUNT ON;
    SET @LevelCode=UPPER(ISNULL(@LevelCode,''));

    SELECT
        R.id AS RuleId,R.RuleName,R.LevelId,L.LevelName,L.LevelCode,
        R.CriticalLabelId,CL.CriticalLabel,CL.ColorCode,CL.Severity,
        R.RulePriority,R.StopProcessing
    FROM dbo.Pms_tbl_CriticalRule R
    INNER JOIN dbo.Pms_tbl_CriticalLevelMaster L ON L.id=R.LevelId
    INNER JOIN dbo.Pms_tbl_CriticalLabel CL ON CL.id=R.CriticalLabelId
    WHERE R.regid=@regid AND R.status=1 AND R.archive=0
      AND L.status=1 AND L.archive=0 AND L.LevelCode=@LevelCode
    ORDER BY R.RulePriority,R.id;

    SELECT
        C.RuleId,C.id AS ConditionId,C.SequenceNo,
        ISNULL(C.ConditionSource,'SELF') AS ConditionSource,C.OpenBracket,
        C.FieldId,F.FieldName,F.FieldKey,F.DataType,F.SourceType,
        C.ChildLevelId,CLV.LevelName AS ChildLevelName,CLV.LevelCode AS ChildLevelCode,
        C.ChildCriticalLabelId,CCL.CriticalLabel AS ChildCriticalLabel,CCL.ColorCode AS ChildCriticalColor,
        C.ChildMeasure,
        C.OperatorId,O.OperatorSymbol,C.CompareValue,C.CompareValue2,
        C.LogicalOperatorId,LO.OperatorName AS LogicalOperator,
        C.CloseBracket,C.GroupNo
    FROM dbo.Pms_tbl_CriticalRuleCondition C
    INNER JOIN dbo.Pms_tbl_CriticalRule R ON R.id=C.RuleId
    INNER JOIN dbo.Pms_tbl_CriticalLevelMaster L ON L.id=R.LevelId
    LEFT JOIN dbo.Pms_tbl_CriticalFieldMaster F ON F.id=C.FieldId
    LEFT JOIN dbo.Pms_tbl_CriticalLevelMaster CLV ON CLV.id=C.ChildLevelId
    LEFT JOIN dbo.Pms_tbl_CriticalLabel CCL ON CCL.id=C.ChildCriticalLabelId
    INNER JOIN dbo.Pms_tbl_CriticalOperatorMaster O ON O.id=C.OperatorId
    LEFT JOIN dbo.Pms_tbl_CriticalLogicalOperator LO ON LO.id=C.LogicalOperatorId
    WHERE R.regid=@regid AND R.status=1 AND R.archive=0
      AND C.status=1 AND C.archive=0 AND L.LevelCode=@LevelCode
    ORDER BY R.RulePriority,C.RuleId,C.SequenceNo;

    /* Legacy action result set retained. */
    SELECT
        RA.RuleId,RA.id AS RuleActionId,RA.ActionId,
        AM.ActionName,AM.ActionCode,RA.ActionValue,RA.RecipientType,RA.SequenceNo
    FROM dbo.Pms_tbl_CriticalRuleAction RA
    INNER JOIN dbo.Pms_tbl_CriticalRule R ON R.id=RA.RuleId
    INNER JOIN dbo.Pms_tbl_CriticalLevelMaster L ON L.id=R.LevelId
    INNER JOIN dbo.Pms_tbl_CriticalActionMaster AM ON AM.id=RA.ActionId
    WHERE R.regid=@regid AND R.status=1 AND R.archive=0
      AND RA.status=1 AND RA.archive=0 AND L.LevelCode=@LevelCode
    ORDER BY R.RulePriority,RA.RuleId,RA.SequenceNo;
END

/* ---------------------------------------------------------
   10. Child summary helper for runtime evaluation.
   IMPORTANT: evaluate bottom-up first so every child has a current
   Pms_tbl_CriticalResult row.

   Example:
       @ParentLevelCode='TASK', @ParentEntityId=100,
       @ChildLevelCode='SUBTASK', @CriticalLabelId=<Critical id>
   --------------------------------------------------------- */
IF OBJECT_ID('dbo.usp_Pms_Critical_GetChildSummary','P') IS NOT NULL
    DROP PROCEDURE dbo.usp_Pms_Critical_GetChildSummary;
GO
CREATE PROCEDURE dbo.usp_Pms_Critical_GetChildSummary
    @regid INT,
    @ParentLevelCode NVARCHAR(30),
    @ParentEntityId INT,
    @ChildLevelCode NVARCHAR(30),
    @CriticalLabelId INT
AS
BEGIN
    SET NOCOUNT ON;

    SET @ParentLevelCode=UPPER(ISNULL(@ParentLevelCode,''));
    SET @ChildLevelCode=UPPER(ISNULL(@ChildLevelCode,''));

    IF @ParentLevelCode NOT IN ('PROJECT','MILESTONE','TASK')
       OR @ChildLevelCode NOT IN ('MILESTONE','TASK','SUBTASK')
    BEGIN
        SELECT 0 AS TotalChildCount,0 AS LabelChildCount,CAST(0 AS DECIMAL(10,2)) AS LabelChildPercent;
        RETURN;
    END

    ;WITH X AS
    (
        SELECT R.EntityId,R.CriticalLabelId
        FROM dbo.Pms_tbl_CriticalResult R
        WHERE R.regid=@regid
          AND R.status=1
          AND R.archive=0
          AND UPPER(R.EntityType)=@ChildLevelCode
          AND
          (
              (@ParentLevelCode='PROJECT' AND R.ProjectId=@ParentEntityId)
              OR (@ParentLevelCode='MILESTONE' AND R.MilestoneId=@ParentEntityId)
              OR (@ParentLevelCode='TASK' AND R.TaskId=@ParentEntityId)
          )
    )
    SELECT
        COUNT(1) AS TotalChildCount,
        SUM(CASE WHEN CriticalLabelId=@CriticalLabelId THEN 1 ELSE 0 END) AS LabelChildCount,
        CAST(
            CASE WHEN COUNT(1)=0 THEN 0
                 ELSE SUM(CASE WHEN CriticalLabelId=@CriticalLabelId THEN 1.0 ELSE 0 END)*100.0/COUNT(1)
            END AS DECIMAL(10,2)
        ) AS LabelChildPercent
    FROM X;
END

/* 6. Save/update rule. */
IF OBJECT_ID('dbo.usp_Pms_CriticalRule_Save','P') IS NOT NULL DROP PROCEDURE dbo.usp_Pms_CriticalRule_Save;
GO
CREATE PROCEDURE dbo.usp_Pms_CriticalRule_Save
    @RuleId INT=0,
    @RuleName NVARCHAR(250),
    @LevelId INT,
    @CriticalLabelId INT=NULL, -- retained only for backward compatibility; UI no longer sends it
    @PriorityId INT=NULL,
    @RulePriority INT=100,
    @StopProcessing BIT=1,
    @Description NVARCHAR(1000)=NULL,
    @regid INT,
    @uid INT=NULL,
    @ConditionsXml XML,
    @ActionsXml XML=NULL
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        IF ISNULL(LTRIM(RTRIM(@RuleName)),'')='' RAISERROR('Rule name is required.',16,1);
        IF NOT EXISTS(SELECT 1 FROM dbo.Pms_tbl_CriticalLevelMaster WHERE id=@LevelId AND status=1 AND archive=0)
            RAISERROR('Invalid Apply On level.',16,1);
        IF @ConditionsXml IS NULL OR @ConditionsXml.exist('/Conditions/Condition')=0
            RAISERROR('At least one IF/ELSE IF condition is required.',16,1);
        IF @ActionsXml IS NULL OR @ActionsXml.exist('/Actions/Action')=0
            RAISERROR('At least one action is required.',16,1);

        /* Validate branch types. */
        IF EXISTS
        (
            SELECT 1 FROM @ConditionsXml.nodes('/Conditions/Condition') T(C)
            WHERE UPPER(T.C.value('@BranchType','NVARCHAR(20)')) NOT IN ('IF','ELSEIF')
        )
            RAISERROR('Conditions are allowed only for IF and ELSE IF branches.',16,1);

        IF EXISTS
        (
            SELECT 1 FROM @ActionsXml.nodes('/Actions/Action') T(C)
            WHERE UPPER(T.C.value('@BranchType','NVARCHAR(20)')) NOT IN ('IF','ELSEIF','ELSE')
        )
            RAISERROR('Invalid action branch type.',16,1);

        DECLARE @HeaderCriticalLabelId INT=NULL;
        DECLARE @SetLabelActionId INT=(SELECT TOP 1 id FROM dbo.Pms_tbl_CriticalActionMaster WHERE ActionCode='SET_CRITICAL_LABEL' AND status=1 AND archive=0 ORDER BY id);

        SELECT TOP 1 @HeaderCriticalLabelId=TRY_CONVERT(INT,NULLIF(T.C.value('@ActionValue','NVARCHAR(1000)'),''))
        FROM @ActionsXml.nodes('/Actions/Action') T(C)
        WHERE T.C.value('@ActionId','INT')=@SetLabelActionId
        ORDER BY T.C.value('@BranchNo','INT'),T.C.value('@SequenceNo','INT');

        IF @HeaderCriticalLabelId IS NOT NULL
           AND NOT EXISTS(SELECT 1 FROM dbo.Pms_tbl_CriticalLabel WHERE id=@HeaderCriticalLabelId AND regid=@regid AND status=1 AND archive=0)
            RAISERROR('Invalid Critical Label selected in action.',16,1);

        IF ISNULL(@RuleId,0)=0
        BEGIN
            INSERT dbo.Pms_tbl_CriticalRule
            (RuleName,LevelId,CriticalLabelId,PriorityId,RulePriority,StopProcessing,Description,regid,uid,status,archive)
            VALUES
            (@RuleName,@LevelId,@HeaderCriticalLabelId,@PriorityId,ISNULL(@RulePriority,100),ISNULL(@StopProcessing,1),@Description,@regid,@uid,1,0);
            SET @RuleId=CAST(SCOPE_IDENTITY() AS INT);
        END
        ELSE
        BEGIN
            UPDATE dbo.Pms_tbl_CriticalRule
            SET RuleName=@RuleName,LevelId=@LevelId,CriticalLabelId=@HeaderCriticalLabelId,
                PriorityId=@PriorityId,RulePriority=ISNULL(@RulePriority,100),StopProcessing=ISNULL(@StopProcessing,1),
                Description=@Description,uid=@uid,modifieddate=GETDATE()
            WHERE id=@RuleId AND regid=@regid AND archive=0;

            IF @@ROWCOUNT=0 RAISERROR('Critical rule not found.',16,1);
            DELETE dbo.Pms_tbl_CriticalRuleCondition WHERE RuleId=@RuleId;
            DELETE dbo.Pms_tbl_CriticalRuleAction WHERE RuleId=@RuleId;
        END

        INSERT dbo.Pms_tbl_CriticalRuleCondition
        (
            RuleId,BranchNo,BranchType,SequenceNo,OpenBracket,ConditionSource,FieldId,ChildLevelId,
            ChildCriticalLabelId,ChildMeasure,OperatorId,CompareValue,CompareValue2,LogicalOperatorId,
            CloseBracket,GroupNo,regid,uid,status,archive
        )
        SELECT
            @RuleId,
            ISNULL(TRY_CONVERT(INT,NULLIF(T.C.value('@BranchNo','NVARCHAR(20)'),'')),1),
            UPPER(ISNULL(NULLIF(T.C.value('@BranchType','NVARCHAR(20)'),''),'IF')),
            T.C.value('@SequenceNo','INT'),
            NULLIF(T.C.value('@OpenBracket','NVARCHAR(10)'),'') ,
            UPPER(ISNULL(NULLIF(T.C.value('@ConditionSource','NVARCHAR(20)'),''),'SELF')),
            TRY_CONVERT(INT,NULLIF(T.C.value('@FieldId','NVARCHAR(20)'),'')),
            TRY_CONVERT(INT,NULLIF(T.C.value('@ChildLevelId','NVARCHAR(20)'),'')),
            TRY_CONVERT(INT,NULLIF(T.C.value('@ChildCriticalLabelId','NVARCHAR(20)'),'')),
            NULLIF(T.C.value('@ChildMeasure','NVARCHAR(30)'),''),
            T.C.value('@OperatorId','INT'),
            NULLIF(T.C.value('@CompareValue','NVARCHAR(500)'),'') ,
            NULLIF(T.C.value('@CompareValue2','NVARCHAR(500)'),'') ,
            TRY_CONVERT(INT,NULLIF(T.C.value('@LogicalOperatorId','NVARCHAR(20)'),'')),
            NULLIF(T.C.value('@CloseBracket','NVARCHAR(10)'),'') ,
            ISNULL(TRY_CONVERT(INT,NULLIF(T.C.value('@GroupNo','NVARCHAR(20)'),'')),1),
            @regid,@uid,1,0
        FROM @ConditionsXml.nodes('/Conditions/Condition') T(C);

        INSERT dbo.Pms_tbl_CriticalRuleAction
        (
            RuleId,ActionId,ActionValue,RecipientType,RecipientValue,SequenceNo,BranchNo,BranchType,regid,uid,status,archive
        )
        SELECT
            @RuleId,
            T.C.value('@ActionId','INT'),
            NULLIF(T.C.value('@ActionValue','NVARCHAR(1000)'),'') ,
            NULLIF(T.C.value('@RecipientType','NVARCHAR(100)'),'') ,
            NULLIF(T.C.value('@RecipientValue','NVARCHAR(1000)'),'') ,
            ISNULL(TRY_CONVERT(INT,NULLIF(T.C.value('@SequenceNo','NVARCHAR(20)'),'')),1),
            ISNULL(TRY_CONVERT(INT,NULLIF(T.C.value('@BranchNo','NVARCHAR(20)'),'')),1),
            UPPER(ISNULL(NULLIF(T.C.value('@BranchType','NVARCHAR(20)'),''),'IF')),
            @regid,@uid,1,0
        FROM @ActionsXml.nodes('/Actions/Action') T(C);

        COMMIT;
        SELECT @RuleId AS RuleId,'Rule saved successfully.' AS Message;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT>0 ROLLBACK;
        DECLARE @Err NVARCHAR(4000)=ERROR_MESSAGE();
        RAISERROR(@Err,16,1);
    END CATCH
END
GO

/* 7. Rule list - Critical Label is derived from Set Critical Label action. */
IF OBJECT_ID('dbo.usp_Pms_CriticalRule_List','P') IS NOT NULL DROP PROCEDURE dbo.usp_Pms_CriticalRule_List;
GO
CREATE PROCEDURE dbo.usp_Pms_CriticalRule_List
    @regid INT,
    @LevelId INT=0
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        R.id,R.RuleName,R.LevelId,L.LevelName,L.LevelCode,
        COALESCE(LA.LabelId,R.CriticalLabelId) AS CriticalLabelId,
        CL.CriticalLabel,CL.ColorCode,CL.Severity,
        R.RulePriority,R.StopProcessing,R.Description,R.status,R.cdate,
        (SELECT COUNT(1) FROM dbo.Pms_tbl_CriticalRuleCondition C WHERE C.RuleId=R.id AND C.status=1 AND C.archive=0) AS ConditionCount,
        (SELECT COUNT(1) FROM dbo.Pms_tbl_CriticalRuleCondition C WHERE C.RuleId=R.id AND C.status=1 AND C.archive=0 AND C.ConditionSource='CHILD') AS ChildConditionCount,
        (SELECT COUNT(1) FROM dbo.Pms_tbl_CriticalRuleAction A WHERE A.RuleId=R.id AND A.status=1 AND A.archive=0) AS ActionCount
    FROM dbo.Pms_tbl_CriticalRule R
    INNER JOIN dbo.Pms_tbl_CriticalLevelMaster L ON L.id=R.LevelId
    OUTER APPLY
    (
        SELECT TOP 1 TRY_CONVERT(INT,RA.ActionValue) AS LabelId
        FROM dbo.Pms_tbl_CriticalRuleAction RA
        INNER JOIN dbo.Pms_tbl_CriticalActionMaster AM ON AM.id=RA.ActionId
        WHERE RA.RuleId=R.id AND RA.status=1 AND RA.archive=0 AND AM.ActionCode='SET_CRITICAL_LABEL'
        ORDER BY ISNULL(RA.BranchNo,1),RA.SequenceNo,RA.id
    ) LA
    LEFT JOIN dbo.Pms_tbl_CriticalLabel CL ON CL.id=COALESCE(LA.LabelId,R.CriticalLabelId)
    WHERE R.regid=@regid AND R.archive=0
      AND L.LevelCode IN ('PROJECT','MILESTONE','TASK','SUBTASK')
      AND (ISNULL(@LevelId,0)=0 OR R.LevelId=@LevelId)
    ORDER BY R.RulePriority,R.id;
END
GO

    ORDER BY ISNULL(C.BranchNo,1),C.SequenceNo;

    FROM dbo.Pms_tbl_CriticalRuleAction RA
    INNER JOIN dbo.Pms_tbl_CriticalActionMaster AM ON AM.id=RA.ActionId
